<script>
(function(){
const gender = document.getElementById('gender');
const height = document.getElementById('height');
const heightVal = document.getElementById('heightVal');
const currentWeight = document.getElementById('currentWeight');
const activity = document.getElementById('activity');
const goalWeight = document.getElementById('goalWeight');

// Frame selection
const frameOptions = document.querySelectorAll('.frame-option');
let frame = 'medium';
frameOptions.forEach(opt => {
opt.addEventListener('click', function(){
frameOptions.forEach(o => o.classList.remove('active'));
this.classList.add('active');
frame = this.dataset.frame;
updateAll();
});
});

// Display elements
const idealWeightSpan = document.getElementById('idealWeight');
const frameText = document.getElementById('frameText');
const millerSpan = document.getElementById('miller');
const robinsonSpan = document.getElementById('robinson');
const devineSpan = document.getElementById('devine');
const petersonSpan = document.getElementById('peterson');
const weeksToGoal = document.getElementById('weeksToGoal');

function updateAll(){
// Update height badge
heightVal.innerText = height.value + ' cm';

const h = parseFloat(height.value); // cm
const isMale = gender.value === 'male';
const wCurrent = parseFloat(currentWeight.value) || 75;
const wGoal = parseFloat(goalWeight.value) || wCurrent;

// Peterson formula (most accurate)
let peterson;
if(isMale){
peterson = 50 + 0.91 * (h - 152.4);
} else {
peterson = 45.5 + 0.91 * (h - 152.4);
}

// Frame adjustment (±5% based on frame)
let frameAdjust = 1.0;
if(frame === 'small') frameAdjust = 0.95;
if(frame === 'large') frameAdjust = 1.05;

const idealMin = (peterson * frameAdjust * 0.94).toFixed(1);
const idealMax = (peterson * frameAdjust * 1.06).toFixed(1);
const idealCenter = (peterson * frameAdjust).toFixed(1);

idealWeightSpan.innerText = `${idealMin} – ${idealMax} kg`;
frameText.innerText = `${frame.charAt(0).toUpperCase()+frame.slice(1)} frame: ${idealCenter} kg (ideal)`;

// Other formulas for comparison
let miller = isMale ? 56.2 + 1.41*(h-152.4)/2.54 : 53.1 + 1.36*(h-152.4)/2.54;
let robinson = isMale ? 52 + 1.9*(h-152.4)/2.54 : 49 + 1.7*(h-152.4)/2.54;
let devine = isMale ? 50 + 2.3*(h-152.4)/2.54 : 45.5 + 2.3*(h-152.4)/2.54;

millerSpan.innerText = miller.toFixed(1) + ' kg';
robinsonSpan.innerText = robinson.toFixed(1) + ' kg';
devineSpan.innerText = devine.toFixed(1) + ' kg';
petersonSpan.innerText = peterson.toFixed(1) + ' kg';

// Goal projection (simple CICO estimate)
if(wGoal < wCurrent){
const diff = wCurrent - wGoal; // kg to lose
const calorieDeficit = diff * 7700; // total kcal deficit needed
const activityMultiplier = parseFloat(activity.value);
const bmr = isMale ? (10*wCurrent + 6.25*h - 5*30 + 5) : (10*wCurrent + 6.25*h - 5*30 - 161);
const tdee = bmr * activityMultiplier;
const deficitPerDay = tdee - 2000; // assuming 2000 kcal intake
if(deficitPerDay > 0){
const days = calorieDeficit / deficitPerDay;
weeksToGoal.innerText = (days/7).toFixed(1) + ' weeks';
} else {
weeksToGoal.innerText = 'adjust diet';
}
} else {
weeksToGoal.innerText = 'maintenance';
}
}

// Add listeners
[gender, height, currentWeight, activity, goalWeight].forEach(el => el.addEventListener('input', updateAll));
updateAll();
})();
</script>
