} else {
metricDiv.style.display = 'none';
imperialDiv.style.display = 'block';
}

});
});

function calculateFFMI() {
let weightKg, heightM, bodyfatPct;

if(currentUnit === 'metric') {
let weight = parseFloat(document.getElementById('weightMetric').value);
let heightCm = parseFloat(document.getElementById('heightMetric').value);
let bf = parseFloat(document.getElementById('bodyfat').value);
if(isNaN(weight) || isNaN(heightCm) || isNaN(bf)) return;
weightKg = weight;
heightM = heightCm / 100;
bodyfatPct = bf;
} else {
let weightLbs = parseFloat(document.getElementById('weightImp').value);
let heightIn = parseFloat(document.getElementById('heightImp').value);
let bf = parseFloat(document.getElementById('bodyfatImp').value);
if(isNaN(weightLbs) || isNaN(heightIn) || isNaN(bf)) return;
weightKg = weightLbs * 0.453592;
heightM = heightIn * 0.0254;
bodyfatPct = bf;
}

if(weightKg <= 0 || heightM <= 0 || bodyfatPct < 0 || bodyfatPct >= 70) return;

let fatMass = weightKg * (bodyfatPct / 100);
let leanMass = weightKg - fatMass; // kg

let ffmi = leanMass / (heightM * heightM);
// adjusted FFMI (for height deviation, Kouri et al.)
let referenceHeight = 1.8; // 180cm reference
let adjustedFFMI = ffmi + (6.1 * (referenceHeight - heightM));

let displayFFMI = adjustedFFMI.toFixed(1);
let rawFFMI = ffmi.toFixed(1);

let lifter = document.getElementById('lifterType').value;
let interpretation = "";
let extraNote = "";

// Descriptive ranges (common in literature)
if(adjustedFFMI < 17) interpretation = "⚠️ Low muscle mass — focus on resistance training & protein.";
else if(adjustedFFMI < 19) interpretation = "📉 Below average muscle mass. Consistent strength training helps.";
else if(adjustedFFMI < 21) interpretation = "✅ Average / Healthy muscle mass.";
else if(adjustedFFMI < 23) interpretation = "💪 Above average — good lean mass!";
else if(adjustedFFMI < 25) interpretation = "🏆 Excellent muscle development. Natural elite range.";
else if(adjustedFFMI < 27) interpretation = "🔥 Exceptional FFMI. For naturals, this is very rare (possible with genetics).";
else interpretation = "⚡ Extremely high FFMI — typical range for enhanced athletes.";

if(lifter === 'natural' && adjustedFFMI > 25.5) {
extraNote = "⚠️ Natural lifter threshold: above 25-26 is uncommon without performance enhancers. Consider honest evaluation.";
} else if(lifter === 'enhanced' && adjustedFFMI < 20) {
extraNote = "ℹ️ Even for enhanced, building foundation first is key. Focus on progressive overload.";
} else {
extraNote = "📊 FFMI is height-adjusted (reference 1.8m). Gold standard for muscle research.";
}

document.getElementById('ffmiNumber').innerText = displayFFMI;
document.getElementById('ffmiInterpret').innerHTML = interpretation;
document.getElementById('ffmiDetail').innerHTML = `Raw FFMI: ${rawFFMI} kg/m² | Lean Mass: ${leanMass.toFixed(1)} kg | ${extraNote}`;
}

document.getElementById('calcBtn').addEventListener('click', calculateFFMI);
window.addEventListener('load', () => {
// set default calculation
calculateFFMI();
});
</script>
