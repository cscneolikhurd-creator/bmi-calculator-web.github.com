name: Deploy to GitHub Pages

on:
push:
branches: [ main ]

permissions:
contents: read
pages: write
id-token: write

jobs:
build:
runs-on: ubuntu-latest

steps:
- uses: actions/checkout@v4

- name: Replace API Key in JS files
run: |
echo "🔑 Injecting API key..."
sed -i "s/%%GEMINI_API_KEY%%/${{ secrets.GEMINI_API_KEY }}/g" ai-health-tips.js

- name: Setup Pages
uses: actions/configure-pages@v4

- name: Build
uses: actions/upload-pages-artifact@v3
with:
path: '.'

- name: Deploy to GitHub Pages
uses: actions/deploy-pages@v4
