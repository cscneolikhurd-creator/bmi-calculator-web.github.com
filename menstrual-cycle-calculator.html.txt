function calculateCycle() {
const lastPeriod = parseInt(document.getElementById('lastPeriod').value);
const cycleLength = parseInt(document.getElementById('cycleLength').value);
const periodLength = parseInt(document.getElementById('periodLength').value);

if (!lastPeriod || !cycleLength || !periodLength) {
document.getElementById('result').innerText = "Please fill all fields";
return;
}

if (lastPeriod < 1 || lastPeriod > 31) {
document.getElementById('result').innerText = "Please enter a valid day (1-31)";
return;
}

// Calculate next period
let nextPeriod = lastPeriod + cycleLength;
let nextPeriodMonth = nextPeriod > 31 ? nextPeriod - 31 : nextPeriod;

// Calculate ovulation (typically 14 days before next period)
const ovulationDay = lastPeriod + (cycleLength - 14);
let ovulationDisplay = ovulationDay > 31 ? ovulationDay - 31 : ovulationDay;

// Fertile window (5 days before ovulation to day of ovulation)
const fertileStart = ovulationDay - 5;
const fertileEnd = ovulationDay;

let fertileStartDisplay = fertileStart > 31 ? fertileStart - 31 : fertileStart;
let fertileEndDisplay = fertileEnd > 31 ? fertileEnd - 31 : fertileEnd;

// Safe days (simplified)
const safeDays = `Days 1-${fertileStart-1} and ${fertileEnd+1}-${cycleLength}`;

const result = `🌸 Your Cycle Predictions:\n\n` +
`Next Period: Day ${nextPeriodMonth} (of next month)\n` +
`Ovulation Day: Day ${ovulationDisplay}\n` +
`Fertile Window: Days ${fertileStartDisplay} - ${fertileEndDisplay}\n\n` +
`Cycle Phases:\n` +
`• Menstruation: Days 1-${periodLength}\n` +
`• Follicular Phase: Days ${periodLength+1}-${ovulationDay-1}\n` +
`• Ovulation: Day ${ovulationDisplay}\n` +
`• Luteal Phase: Days ${ovulationDay+1}-${cycleLength}\n\n` +
`💡 Note: This is an estimate. Actual cycles may vary.`;

document.getElementById('result').innerText = result;
}
