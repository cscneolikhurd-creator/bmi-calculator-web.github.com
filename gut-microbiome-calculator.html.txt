function calculateGutScore() {
// Get all input values
const fiber = parseInt(document.getElementById('fiber').value);
const probiotic = parseInt(document.getElementById('probiotic').value);
const prebiotic = parseInt(document.getElementById('prebiotic').value);
const processed = parseInt(document.getElementById('processed').value);
const alcohol = parseInt(document.getElementById('alcohol').value);
const antibiotics = parseInt(document.getElementById('antibiotics').value);
const sleep = parseInt(document.getElementById('sleep').value);
const exercise = parseInt(document.getElementById('exercise').value);
const stress = parseInt(document.getElementById('stress').value);

// Count symptoms (each symptom reduces score)
let symptoms = 0;
const symptomIds = ['symptom1', 'symptom2', 'symptom3', 'symptom4', 'symptom5', 'symptom6'];
symptomIds.forEach(id => {
if (document.getElementById(id).checked) symptoms++;
});

// Calculate base score (max 100)
let score = 0;
let maxScore = 44; // Total possible points from factors (excluding symptoms)

// Add up all positive factors (each out of 4)
score += fiber * 3; // Fiber is weighted more (max 12)
score += probiotic * 2; // Probiotics (max 8)
score += prebiotic * 2; // Prebiotics (max 8)
score += processed * 1; // Processed foods reverse (max 4)
score += alcohol * 1; // Alcohol (max 4)
score += antibiotics * 1; // Antibiotics (max 4)
score += sleep * 2; // Sleep weighted (max 8)
score += exercise * 2; // Exercise weighted (max 8)
score += (4 - stress) * 2; // Stress (inverse, max 8)

// Subtract for symptoms (each symptom -2 points)
score -= symptoms * 2;

// Calculate max possible (44 from factors - 0 symptoms = 44)
// Scale to 0-100
const scaledScore = Math.max(0, Math.min(100, Math.round((score / 44) * 100)));

// Update score display
document.getElementById('scoreNumber').innerText = scaledScore;
document.getElementById('scoreMeter').style.width = scaledScore + '%';

// Score category
let category = '';
if (scaledScore >= 80) category = 'Excellent';
else if (scaledScore >= 60) category = 'Good';
else if (scaledScore >= 40) category = 'Fair';
else if (scaledScore >= 20) category = 'Poor';
else category = 'Very Poor';

document.getElementById('scoreCategory').innerText = category;

// Update individual scores display
document.getElementById('probioticScore').innerText = probiotic + '/4';
document.getElementById('probioticBar').style.width = (probiotic * 25) + '%';

document.getElementById('prebioticScore').innerText = prebiotic + '/4';
document.getElementById('prebioticBar').style.width = (prebiotic * 25) + '%';

document.getElementById('processedScore').innerText = processed + '/4';
document.getElementById('processedBar').style.width = (processed * 25) + '%';

// Fiber display
const fiberText = ['<10g', '10-15g', '15-25g', '25-35g', '>35g'];
document.getElementById('fiberScore').innerText = fiberText[fiber];
if (fiber >= 3) {
document.getElementById('fiberDesc').innerText = 'Optimal!';
} else {
document.getElementById('fiberDesc').innerText = 'Target: 30g+';
}

// Diversity score (based on probiotic + prebiotic + fiber)
const diversity = Math.round((probiotic * 2 + prebiotic * 2 + fiber * 3) / 28 * 100);
document.getElementById('diversityScore').innerText = diversity + '/100';
if (diversity >= 70) {
document.getElementById('diversityDesc').innerText = 'Good diversity';
} else if (diversity >= 50) {
document.getElementById('diversityDesc').innerText = 'Moderate diversity';
} else {
document.getElementById('diversityDesc').innerText = 'Low diversity';
}

// Gut-brain axis text
let gutBrainText = '';
if (stress <= 1) {
gutBrainText = 'Your stress levels are high, which can negatively impact gut health. Consider meditation, deep breathing, or yoga.';
} else if (sleep <= 1) {
gutBrainText = 'Poor sleep affects the gut microbiome. Aim for 7-8 hours of quality sleep.';
} else if (scaledScore >= 70) {
gutBrainText = 'Your gut-brain axis appears balanced. Continue your healthy habits!';
} else {
gutBrainText = 'Your stress and lifestyle factors may be affecting your gut health. Focus on the recommendations below.';
}
document.getElementById('gutBrainText').innerText = gutBrainText;

// Generate recommendations
generateRecommendations(fiber, probiotic, prebiotic, processed, stress, sleep, exercise, symptoms);
}
