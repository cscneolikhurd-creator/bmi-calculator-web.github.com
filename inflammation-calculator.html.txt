<script>
function calculateInflammation() {
// Get all values
const omega = parseInt(document.getElementById('omegaRatio').value);
const sugar = parseInt(document.getElementById('sugar').value);
const alcohol = parseInt(document.getElementById('alcohol').value);
const veggies = parseInt(document.getElementById('veggies').value);
const exercise = parseInt(document.getElementById('exercise').value);
const bmi = parseInt(document.getElementById('bmi').value);
const smoking = parseInt(document.getElementById('smoking').value);
const sleep = parseInt(document.getElementById('sleep').value);

// Count symptoms
let symptoms = 0;
for (let i = 1; i <= 6; i++) {
if (document.getElementById(`symptom${i}`).checked) symptoms++;
}

// Calculate scores
let dietScore = (omega * 8) + (sugar * 6) + (alcohol * 5) + (veggies * 6);
let lifestyleScore = (exercise * 6) + (bmi * 8) + (smoking * 8) + (sleep * 5) + (symptoms * 4);

let totalScore = dietScore + lifestyleScore;
let maxScore = 240;
let percentage = Math.round((totalScore / maxScore) * 100);

// Update display
document.getElementById('scoreNumber').innerText = percentage;
document.getElementById('scoreMeter').style.width = percentage + '%';

// Category
let category = '';
if (percentage < 25) category = 'Low Inflammation';
else if (percentage < 50) category = 'Moderate Inflammation';
else if (percentage < 75) category = 'High Inflammation';
else category = 'Severe Inflammation';
document.getElementById('scoreCategory').innerText = category;

// CRP estimate
let crp = (percentage / 20).toFixed(1);
document.getElementById('crpValue').innerText = crp + ' mg/L';

let crpRisk = '';
if (crp < 1) crpRisk = 'Low Risk';
else if (crp < 3) crpRisk = 'Moderate Risk';
else crpRisk = 'High Risk';
document.getElementById('crpRisk').innerText = crpRisk;

// Diet/Lifestyle scores
document.getElementById('dietScore').innerText = dietScore + '/100';
document.getElementById('lifestyleScore').innerText = lifestyleScore + '/140';

// Recommendations
let recs = [];
if (omega > 1) recs.push({icon:'🐟', title:'Improve Omega-3:6 Ratio', desc:'Eat fatty fish 2x/week, add flaxseeds, reduce vegetable oils'});
if (sugar > 2) recs.push({icon:'🍬', title:'Reduce Added Sugar', desc:'Cut sugary drinks, desserts, and processed foods'});
if (veggies < 2) recs.push({icon:'🥗', title:'Eat More Vegetables', desc:'Aim for 7+ servings of colorful veggies daily'});
if (exercise > 1) recs.push({icon:'🏃', title:'Increase Exercise', desc:'150 minutes moderate exercise weekly reduces inflammation'});
if (bmi > 1) recs.push({icon:'⚖️', title:'Healthy Weight', desc:'Losing 5-10% body weight reduces inflammatory markers'});
if (smoking > 1) recs.push({icon:'🚭', title:'Quit Smoking', desc:'Smoking increases CRP by 30-50%'});
if (sleep > 2) recs.push({icon:'😴', title:'Improve Sleep', desc:'Poor sleep increases inflammatory cytokines'});

document.getElementById('recommendations').innerHTML = recs.slice(0,4).map(r => `
<div class="rec-item">
<div class="rec-icon">${r.icon}</div>
<div><div class="rec-title">${r.title}</div><div class="rec-desc">${r.desc}</div></div>
</div>
`).join('');
}

function shareResults() {
navigator.clipboard.writeText(window.location.href);
alert('Link copied!');
}

window.onload = calculateInflammation;
</script>
