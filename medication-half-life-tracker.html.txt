<script>
let chart = null;

const presetDrugs = {
ibuprofen: { half: 2, dose: 400 },
caffeine: { half: 5, dose: 200 },
loratadine: { half: 8, dose: 10 },
omeprazole: { half: 1.5, dose: 20 },
diazepam: { half: 48, dose: 10 }
};

document.querySelectorAll('.preset-btn').forEach(btn => {
btn.addEventListener('click', () => {
const drug = btn.getAttribute('data-drug');
const data = presetDrugs[drug];
if(data) {
document.getElementById('medName').value = btn.getAttribute('data-drug') || drug;
document.getElementById('halfLife').value = data.half;
document.getElementById('dose').value = data.dose;
calculate();
}
});
});

function calculate() {
let halfLife = parseFloat(document.getElementById('halfLife').value);
let dose = parseFloat(document.getElementById('dose').value);
let hoursSince = parseFloat(document.getElementById('hoursSince').value);
let medName = document.getElementById('medName').value || 'Medication';

if(isNaN(halfLife) || halfLife <= 0 || isNaN(dose) || dose <= 0 || isNaN(hoursSince) || hoursSince < 0) {
return;
}

// Elimination constant k = ln(2)/halfLife
let k = Math.LN2 / halfLife;
let initialConc = dose;
let currentConc = dose * Math.exp(-k * hoursSince);
let remainingPct = (currentConc / dose) * 100;

// Time to 95% elimination (3.3 half-lives)
let elimTime95 = halfLife * 4.32; // 95% elimination = 4.32 half-lives

document.getElementById('initialConc').innerHTML = initialConc + ' mg';
document.getElementById('currentConc').innerHTML = currentConc.toFixed(2) + ' mg';
document.getElementById('remainingPct').innerHTML = remainingPct.toFixed(1) + '%';
document.getElementById('elimTime').innerHTML = elimTime95.toFixed(1) + ' hours';

// Interpretation
let interpretation = '';
if(remainingPct > 50) {
interpretation = `⚠️ ${Math.round(remainingPct)}% of ${medName} remains in your system. Peak effects still present.`;
} else if(remainingPct > 25) {
interpretation = `ℹ️ ${Math.round(remainingPct)}% remains. Therapeutic effects are decreasing.`;
} else if(remainingPct > 10) {
interpretation = `✅ ${Math.round(remainingPct)}% remains. Most of the drug has been eliminated.`;
} else {
interpretation = `✅ Less than 10% remains. ${medName} is mostly cleared from your system.`;
}

interpretation += `<br><br>📊 ${medName} has a half-life of ${halfLife} hours. It takes about ${(halfLife * 4.32).toFixed(1)} hours (${(halfLife * 4.32 / 24).toFixed(1)} days) for 95% elimination.`;

document.getElementById('interpretation').innerHTML = interpretation;

// Generate chart data (48 hours or 4 half-lives)
let maxHours = Math.max(48, halfLife * 5);
let points = [];
let timePoints = [];

for(let i = 0; i <= 100; i++) {
let t = (i / 100) * maxHours;
let conc = dose * Math.exp(-k * t);
points.push(conc);
if(i % 10 === 0) timePoints.push(t.toFixed(1) + 'h');
else timePoints.push('');
}

let labels = [];
for(let i = 0; i <= 100; i++) {
if(i % 10 === 0) labels.push(((i / 100) * maxHours).toFixed(1) + 'h');
else labels.push('');
}

if(chart) {
chart.destroy();
}

const ctx = document.getElementById('concentrationChart').getContext('2d');
chart = new Chart(ctx, {
type: 'line',
data: {
labels: labels,
datasets: [{
label: `${medName} Concentration (mg)`,
data: points,
borderColor: '#667eea',
backgroundColor: 'rgba(102, 126, 234, 0.1)',
borderWidth: 3,
fill: true,
tension: 0.4,
pointRadius: 0,
pointHoverRadius: 5
}]
},
options: {
responsive: true,
maintainAspectRatio: true,
plugins: {
legend: {
position: 'top',
},
tooltip: {
callbacks: {
label: function(context) {
return `Concentration: ${context.raw.toFixed(2)} mg`;
}
}
}
},
scales: {
y: {
title: {
display: true,
text: 'Concentration (mg)'
},
beginAtZero: true
},
x: {
title: {
display: true,
text: 'Time (hours)'
}
}
}
}
});
}

document.getElementById('calculateBtn').addEventListener('click', calculate);

// Auto-calculate on input changes
['halfLife', 'dose', 'hoursSince'].forEach(id => {
document.getElementById(id).addEventListener('input', calculate);
});

calculate();
</script>
