function getFitnessLevel(vo2, gender, age) {
// Simplified ACSM categories (mL/kg/min)
let level = '';
if(gender === 'male') {
if(vo2 < 35) level = '🔴 Poor';
else if(vo2 < 42) level = '🟠 Fair';
else if(vo2 < 48) level = '🟡 Average';
else if(vo2 < 54) level = '🟢 Good';
else if(vo2 < 60) level = '💪 Excellent';
else level = '🏆 Superior (Athlete)';
} else {
if(vo2 < 29) level = '🔴 Poor';
else if(vo2 < 35) level = '🟠 Fair';
else if(vo2 < 40) level = '🟡 Average';
else if(vo2 < 46) level = '🟢 Good';
else if(vo2 < 52) level = '💪 Excellent';
else level = '🏆 Superior (Athlete)';
}
return level;
}

function getPercentile(vo2, gender, age) {
if(age < 30) {
if(gender === 'male') {
if(vo2 > 55) return 'Top 5%';
if(vo2 > 48) return 'Top 25%';
if(vo2 > 42) return 'Average (50th)';
return 'Below average';
} else {
if(vo2 > 48) return 'Top 5%';
if(vo2 > 42) return 'Top 25%';
if(vo2 > 36) return 'Average (50th)';
return 'Below average';
}
} else if(age < 45) {
if(gender === 'male') {
if(vo2 > 52) return 'Top 5%';
if(vo2 > 45) return 'Top 25%';
if(vo2 > 39) return 'Average';
return 'Below avg';
} else {
if(vo2 > 45) return 'Top 5%';
if(vo2 > 39) return 'Top 25%';
if(vo2 > 34) return 'Average';
return 'Below avg';
}
} else {
if(gender === 'male') {
if(vo2 > 48) return 'Top 5%';
if(vo2 > 40) return 'Top 25%';
if(vo2 > 34) return 'Average';
return 'Below avg';
} else {
if(vo2 > 42) return 'Top 5%';
if(vo2 > 36) return 'Top 25%';
if(vo2 > 30) return 'Average';
return 'Below avg';
}
}
}

document.getElementById('calcVO2').addEventListener('click', () => {
let vo2 = 0;

if(currentMethod === 'resting') {
let gender = document.getElementById('genderRest').value;
let age = parseFloat(document.getElementById('ageRest').value);
let hr = parseFloat(document.getElementById('restingHR').value);
let activity = document.getElementById('activityLevel').value;

if(isNaN(age) || isNaN(hr)) return;

// Base formula: 15.3 * (HRmax/HRrest) adjustment
let hrMax = 208 - (0.7 * age);
let hrRatio = hrMax / hr;
let baseVO2 = 45 * hrRatio;

// Activity modifier
let actMod = {sedentary: 0.85, light: 0.92, moderate: 1.0, active: 1.08, very: 1.15};
vo2 = baseVO2 * (actMod[activity] || 1.0);
vo2 = Math.min(85, Math.max(20, vo2));

} else { // Walk test method
let gender = document.getElementById('genderWalk').value;
let age = parseFloat(document.getElementById('ageWalk').value);
let weight = parseFloat(document.getElementById('weightWalk').value);
let time = parseFloat(document.getElementById('walkTime').value);
let hrWalk = parseFloat(document.getElementById('walkHR').value);

if(isNaN(age) || isNaN(weight) || isNaN(time) || isNaN(hrWalk)) return;

// Kline et al. formula for 1-mile walk
let genderVal = (gender === 'male') ? 1 : 0;
vo2 = 132.853 - (0.0769 * weight) - (0.3877 * age) + (6.315 * genderVal) - (3.2649 * time) - (0.1565 * hrWalk);
vo2 = Math.min(75, Math.max(18, vo2));
}

vo2 = Math.round(vo2 * 10) / 10;
document.getElementById('vo2Number').innerHTML = vo2 + ' mL/kg/min';

let genderCurr = currentMethod === 'resting' ? document.getElementById('genderRest').value : document.getElementById('genderWalk').value;
let ageCurr = currentMethod === 'resting' ? parseFloat(document.getElementById('ageRest').value) : parseFloat(document.getElementById('ageWalk').value);

let fitness = getFitnessLevel(vo2, genderCurr, ageCurr);
let percentile = getPercentile(vo2, genderCurr, ageCurr);

document.getElementById('fitnessCategory').innerHTML = fitness;
document.getElementById('percentileInfo').innerHTML = `📊 ${percentile} for age ${Math.round(ageCurr)}`;

let tip = '';
if(vo2 < 35) tip = '💡 Start with brisk walking 30min/day, 3-4 times weekly. Gradually increase intensity.';
else if(vo2 < 45) tip = '💡 Add interval training (e.g., 1 min fast, 2 min slow). Consistency is key.';
else if(vo2 < 55) tip = '💡 Great base! Try tempo runs or HIIT to break plateau.';
else tip = '🏅 Excellent endurance! Consider periodized training & VO2 max specific intervals (3-5 min hard efforts).';

document.getElementById('trainingTip').innerHTML = tip;
});
