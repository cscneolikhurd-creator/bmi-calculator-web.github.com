function calculateCircadian() {
// Calculate MEQ score
let meqScore = 0;
for (let i = 1; i <= 5; i++) {
const selected = document.querySelector(`input[name="q${i}"]:checked`);
if (selected) {
meqScore += parseInt(selected.value);
}
}

// Determine chronotype
let chronotype = '';
let icon = '';
let desc = '';

if (meqScore >= 22 && meqScore <= 30) {
chronotype = 'Lion (Definite Morning)';
icon = '🦁';
desc = 'You're an early bird. Most productive in early morning, best to sleep early.';
} else if (meqScore >= 18 && meqScore <= 21) {
chronotype = 'Bear (Moderate Morning)';
icon = '🐻';
desc = 'You follow the sun. Most productive mid-morning, sleep 11 PM - 7 AM.';
} else if (meqScore >= 12 && meqScore <= 17) {
chronotype = 'Wolf (Intermediate)';
icon = '🐺';
desc = 'You're a night owl. Most productive afternoon/evening, prefer late sleep.';
} else if (meqScore >= 5 && meqScore <= 11) {
chronotype = 'Dolphin (Definite Evening)';
icon = '🐬';
desc = 'Light sleeper, irregular pattern. Need consistent routine most of all.';
} else {
chronotype = 'Bear';
icon = '🐻';
desc = 'You follow the sun. Most productive mid-morning.';
}

// Update chronotype display
document.getElementById('chronotypeIcon').innerText = icon;
document.getElementById('chronotypeName').innerText = chronotype;
document.getElementById('chronotypeDesc').innerText = desc;
document.getElementById('meqScore').innerText = meqScore;

// Chronotype range
let range = '';
if (meqScore >= 22) range = 'Definite Morning';
else if (meqScore >= 18) range = 'Moderate Morning';
else if (meqScore >= 12) range = 'Intermediate';
else range = 'Definite Evening';
document.getElementById('chronotypeRange').innerText = range;

// Calculate social jetlag
function timeToMinutes(timeStr) {
const [hours, minutes] = timeStr.split(':').map(Number);
return hours * 60 + minutes;
}

const weekdayWake = timeToMinutes(document.getElementById('weekdayWake').value);
const weekdayBed = timeToMinutes(document.getElementById('weekdayBed').value);
const weekendWake = timeToMinutes(document.getElementById('weekendWake').value);
const weekendBed = timeToMinutes(document.getElementById('weekendBed').value);

// Adjust for overnight (if bed time is after midnight)
const weekdaySleep = weekdayBed > weekdayWake ? weekdayBed - (24*60) : weekdayBed;
const weekendSleep = weekendBed > weekendWake ? weekendBed - (24*60) : weekendBed;

const weekdayDuration = weekdayWake - weekdaySleep;
const weekendDuration = weekendWake - weekendSleep;

const jetlag = Math.abs(weekendWake - weekdayWake) / 60;
const roundedJetlag = Math.round(jetlag * 10) / 10;

document.getElementById('jetlagHours').innerText = roundedJetlag;

// Jetlag bar (max 4 hours)
const jetlagPercent = Math.min(100, (roundedJetlag / 4) * 100);
document.getElementById('jetlagBar').style.width = jetlagPercent + '%';

let jetlagDesc = '';
if (roundedJetlag < 1) {
jetlagDesc = 'Excellent! Minimal social jetlag.';
document.getElementById('jetlagCard').style.background = 'linear-gradient(145deg, #dcfce7, #bbf7d0)';
} else if (roundedJetlag < 2) {
jetlagDesc = 'Good. Small circadian disruption.';
document.getElementById('jetlagCard').style.background = 'linear-gradient(145deg, #fef9c3, #fde047)';
} else if (roundedJetlag < 3) {
jetlagDesc = 'Moderate disruption. Try to keep consistent times.';
document.getElementById('jetlagCard').style.background = 'linear-gradient(145deg, #fee2e2, #fecaca)';
} else {
jetlagDesc = 'Severe social jetlag. This can impact health significantly.';
document.getElementById('jetlagCard').style.background = 'linear-gradient(145deg, #fecaca, #fca5a5)';
}
document.getElementById('jetlagDesc').innerText = jetlagDesc;

// Light exposure values
const morningLight = document.getElementById('morningLight').value;
document.getElementById('morningLightValue').innerText = morningLight + ' min';

const screenTime = document.getElementById('screenTime').value;
document.getElementById('screenTimeValue').innerText = screenTime + ' hr';

// Generate optimal schedule based on chronotype
generateSchedule(chronotype);

// Generate tips
generateTips(chronotype, morningLight, screenTime, roundedJetlag);
}
