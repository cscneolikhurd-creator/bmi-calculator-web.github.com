<script>
let selectedFrame = null;
let selectedFat = null;
let selectedMuscle = null;
let selectedMetab = null;

// Option selection handlers
function setupOptionButtons(className, storageVar, optionsId) {
document.querySelectorAll(optionsId + ' .option-btn').forEach(btn => {
btn.addEventListener('click', () => {
document.querySelectorAll(optionsId + ' .option-btn').forEach(b => b.classList.remove('selected'));
btn.classList.add('selected');
if(optionsId === '#frameOptions') selectedFrame = btn.getAttribute('data-frame');
if(optionsId === '#fatOptions') selectedFat = btn.getAttribute('data-fat');
if(optionsId === '#muscleOptions') selectedMuscle = btn.getAttribute('data-muscle');
if(optionsId === '#metabolismOptions') selectedMetab = btn.getAttribute('data-metab');
});
});
}

setupOptionButtons('frame', 'selectedFrame', '#frameOptions');
setupOptionButtons('fat', 'selectedFat', '#fatOptions');
setupOptionButtons('muscle', 'selectedMuscle', '#muscleOptions');
setupOptionButtons('metab', 'selectedMetab', '#metabolismOptions');

function calculateSomatotype() {
let endo = 0, meso = 0, ecto = 0;

// Body fat tendency -> Endomorph
if(selectedFat === 'high') endo += 4;
else if(selectedFat === 'moderate') endo += 2;
else endo += 0;

// Metabolism -> Endomorph
if(selectedMetab === 'slow') endo += 3;
else if(selectedMetab === 'normal') endo += 1;
else endo += 0;

// Frame -> Mesomorph
if(selectedFrame === 'large') meso += 4;
else if(selectedFrame === 'medium') meso += 2;
else meso += 0;

// Muscle building -> Mesomorph
if(selectedMuscle === 'high') meso += 4;
else if(selectedMuscle === 'medium') meso += 2;
else meso += 0;

// Frame small -> Ectomorph
if(selectedFrame === 'small') ecto += 4;

// Metabolism fast -> Ectomorph
if(selectedMetab === 'fast') ecto += 4;
else if(selectedMetab === 'normal') ecto += 1;

// Muscle low -> Ectomorph
if(selectedMuscle === 'low') ecto += 3;

// Wrist size adjustment
let wrist = parseFloat(document.getElementById('wristSize').value);
if(!isNaN(wrist) && wrist > 0) {
if(wrist < 16) ecto += 2;
else if(wrist > 19) endo += 2;
else meso += 1;
}

// Normalize to 1-7 scale
let total = endo + meso + ecto;
if(total > 0) {
endo = (endo / total) * 7;
meso = (meso / total) * 7;
ecto = (ecto / total) * 7;
}

// Determine dominant type
let maxVal = Math.max(endo, meso, ecto);
let somaType = '';
if(maxVal === endo && endo > meso && endo > ecto) somaType = 'Endomorph';
else if(maxVal === meso && meso > endo && meso > ecto) somaType = 'Mesomorph';
else if(maxVal === ecto && ecto > endo && ecto > meso) somaType = 'Ectomorph';
else if(endo > 4 && meso > 3) somaType = 'Endo-Mesomorph';
else if(ecto > 4 && meso > 3) somaType = 'Ecto-Mesomorph';
else somaType = 'Mixed Type';

return { endo, meso, ecto, somaType };
}

const recommendations = {

tips: '💡 You\'re genetically gifted! Vary your routine every 6-8 weeks to avoid plateaus.'
},
'Ectomorph': {

document.getElementById('analyzeBtn').addEventListener('click', () => {
if(!selectedFrame || !selectedFat || !selectedMuscle || !selectedMetab) {
alert('Please answer all questions to analyze your body type.');
return;
}

const result = calculateSomatotype();
const rec = getRecommendations(result.somaType);

document.getElementById('resultBox').style.display = 'block';
const somaElement = document.getElementById('somatype');
somaElement.textContent = result.somaType;
somaElement.className = 'somatype';

if(result.somaType.includes('Endomorph')) somaElement.classList.add('endomorph');
else if(result.somaType.includes('Mesomorph')) somaElement.classList.add('mesomorph');
else if(result.somaType.includes('Ectomorph')) somaElement.classList.add('ectomorph');

document.getElementById('scores').innerHTML = `Endomorph: ${result.endo.toFixed(1)} | Mesomorph: ${result.meso.toFixed(1)} | Ectomorph: ${result.ecto.toFixed(1)}`;

document.getElementById('recommendation').innerHTML = `
${rec.diet}<br><br>
${rec.workout}<br><br>
${rec.tips}
`;

});
